Programmatic access for scripts and tools. All endpoints require a bearer token.

Quick start

1. Mint a token at /user/tokens. 2. Send it in the Authorization header.

curl

TOKEN="dikarya_..."

# Identity check
curl -H "Authorization: Bearer $TOKEN" https://dikarya.us/api/v1/me

# Create a job
curl -X POST https://dikarya.us/api/v1/jobs \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
        "input_type": "fasta",
        "sequence": ">A\nACGT...\n>B\nACGT...",
        "tree_method": "fasttree",
        "alignment_method": "mafft",
        "notes": "remote run"
      }'

# Poll
curl -H "Authorization: Bearer $TOKEN" https://dikarya.us/api/v1/jobs/<id>

# Download the tree
curl -H "Authorization: Bearer $TOKEN" \
  https://dikarya.us/api/v1/jobs/<id>/files/tree.newick -o tree.newick

Python

import requests, time

API = "https://dikarya.us/api/v1"
HEADERS = {"Authorization": "Bearer dikarya_..."}

r = requests.post(f"{API}/jobs",
    headers={**HEADERS, "Idempotency-Key": "my-run-2026-05-11"},
    json={"input_type": "fasta", "sequence": open("seqs.fasta").read(),
          "tree_method": "fasttree"})
r.raise_for_status()
job_id = r.json()["data"]["id"]

while True:
    job = requests.get(f"{API}/jobs/{job_id}", headers=HEADERS).json()["data"]
    if job["status"] in ("completed", "failed", "error"):
        break
    time.sleep(5)

tree = requests.get(f"{API}/jobs/{job_id}/files/tree.newick", headers=HEADERS).text
print(tree)
