Quick start

1. Mint a token at /user/tokens. 2. Send it in the Authorization header.

curl

Requires jq. Use "input_type": "pasted_sequence" when sending FASTA text in JSON — "fasta" is reserved for file-upload flows and is not supported on this endpoint. 

TOKEN="dikarya_..."

# Identity check
curl -H "Authorization: Bearer $TOKEN" https://dikarya.us/api/v1/me

# Create a small working tree job and save the returned UUID
JOB_ID=$(
  curl -s -X POST https://dikarya.us/api/v1/jobs \
    -H "Authorization: Bearer $TOKEN" \
    -H "Content-Type: application/json" \
    -d '{
      "input_type": "pasted_sequence",
      "sequence": ">Sample_A\nATGCGTACGTAGCTAGCTAGCTAGCTAGCTAACGATCGATCGATCGATCGATCGATCGATCGATCGATCGATCGATCGATCGATCGATCGATCGATCG\n>Sample_B\nATGCGTACGTAGCTAGCTAGCTAGCTAGCTAACGATCGATCGATCGATCGATCGATCGATCGATCGATCGATCGATCGATCGTTTGATCGATCGATCG\n>Sample_C\nATGCGTACGTAGCTAGCTAGCTAGCTAGCTAACGATCGATCGATCGATCGATCGATCGATCGATCGATCGATCGATCGATCGATCGTTTGATCG\n>Sample_D\nATGCGTACGTAGCTAGCTAGCTAGCTAGCTAACGATCGATCGATCGATCGATCGATCGATCGATCGATCGATCGTTTGATCGATCGATCGATCGATCG",
      "tree_method": "fasttree",
      "alignment_method": "mafft",
      "trimming_method": "none",
      "notes": "API test with valid pasted FASTA"
    }' | jq -r '.data.id'
)

echo "$JOB_ID"

# Poll until the job finishes. Trees can take far longer than two minutes,
# so the deadline is configurable rather than a fixed 60 iterations.
MAX_WAIT_SECONDS="${MAX_WAIT_SECONDS:-3600}"
STATUS=""
deadline=$((SECONDS + MAX_WAIT_SECONDS))
while (( SECONDS < deadline )); do
  STATUS=$(
    curl -s -H "Authorization: Bearer $TOKEN" \
      "https://dikarya.us/api/v1/jobs/$JOB_ID" | jq -r '.data.status // .error.message'
  )

  echo "$(date +%T) $STATUS"

  case "$STATUS" in
    completed) break ;;
    failed|error) echo "Job did not complete: $STATUS" >&2; exit 1 ;;
  esac

  sleep 2
done

if [[ "$STATUS" != "completed" ]]; then
  echo "Job did not complete within ${MAX_WAIT_SECONDS}s" >&2
  exit 1
fi

# Show the final job record
curl -s -H "Authorization: Bearer $TOKEN" \
  "https://dikarya.us/api/v1/jobs/$JOB_ID" | jq

# Download the Newick tree. -f so an error response is not saved as tree.newick.
curl -fS -H "Authorization: Bearer $TOKEN" \
  "https://dikarya.us/api/v1/jobs/$JOB_ID/files/tree.newick" \
  -o tree.newick

cat tree.newick

Python

Reads your token from DIKARYA_TOKEN. Same pasted_sequence convention as the curl example. 

import os
import time
import requests

API = "https://dikarya.us/api/v1"
TOKEN = os.environ["DIKARYA_TOKEN"]
HEADERS = {"Authorization": f"Bearer {TOKEN}"}

sequence = """>Sample_A
ATGCGTACGTAGCTAGCTAGCTAGCTAGCTAACGATCGATCGATCGATCGATCGATCGATCGATCGATCGATCGATCGATCGATCGATCGATCGATCG
>Sample_B
ATGCGTACGTAGCTAGCTAGCTAGCTAGCTAACGATCGATCGATCGATCGATCGATCGATCGATCGATCGATCGATCGATCGTTTGATCGATCGATCG
>Sample_C
ATGCGTACGTAGCTAGCTAGCTAGCTAGCTAACGATCGATCGATCGATCGATCGATCGATCGATCGATCGATCGATCGATCGATCGTTTGATCG
>Sample_D
ATGCGTACGTAGCTAGCTAGCTAGCTAGCTAACGATCGATCGATCGATCGATCGATCGATCGATCGATCGATCGTTTGATCGATCGATCGATCGATCG
"""

payload = {
    "input_type": "pasted_sequence",
    "sequence": sequence,
    "tree_method": "fasttree",
    "alignment_method": "mafft",
    "trimming_method": "none",
    "notes": "Python API test with valid pasted FASTA",
}

# (connect, read) timeouts; without them a stalled connection blocks forever.
TIMEOUT = (5, 30)

response = requests.post(
    f"{API}/jobs",
    headers={**HEADERS, "Content-Type": "application/json"},
    json=payload,
    timeout=TIMEOUT,
)
response.raise_for_status()
job_id = response.json()["data"]["id"]
print("Job ID:", job_id)

while True:
    response = requests.get(f"{API}/jobs/{job_id}", headers=HEADERS, timeout=TIMEOUT)
    response.raise_for_status()
    job = response.json()["data"]
    print(job["status"])

    if job["status"] in ("completed", "failed", "error"):
        break

    time.sleep(2)

if job["status"] != "completed":
    raise RuntimeError(f"Job did not complete: {job['status']}")

tree_response = requests.get(
    f"{API}/jobs/{job_id}/files/tree.newick",
    headers=HEADERS,
    timeout=TIMEOUT,
)
tree_response.raise_for_status()
print(tree_response.text)
